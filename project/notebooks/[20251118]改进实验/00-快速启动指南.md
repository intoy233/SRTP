# -*- coding: utf-8 -*-
"""
特征选择实验
目标: 从234维降至50-100维，缓解维度灾难
"""

from pathlib import Path
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestRegressor
from sklearn.linear_model import Lasso
from sklearn.feature_selection import RFE
from src.final_viv_predictor import VIVPredictor

def select_features_rf(X, y, top_k=80):
    """使用随机森林选择top-k重要特征"""
    rf = RandomForestRegressor(n_estimators=100, random_state=42)
    rf.fit(X, y)
    importances = rf.feature_importances_
    indices = np.argsort(importances)[::-1][:top_k]
    return indices

def select_features_lasso(X, y, alpha=0.01):
    """使用Lasso选择非零系数特征"""
    lasso = Lasso(alpha=alpha, max_iter=5000)
    lasso.fit(X, y)
    indices = np.where(np.abs(lasso.coef_) > 1e-5)[0]
    return indices

def evaluate_feature_subset(X, y, selected_indices):
    """评估特征子集的性能"""
    from sklearn.model_selection import cross_val_score
    from sklearn.linear_model import BayesianRidge

    X_selected = X[:, selected_indices]
    model = BayesianRidge()
    scores = cross_val_score(model, X_selected, y, cv=5,
                             scoring='r2', n_jobs=-1)
    return scores.mean(), scores.std()
